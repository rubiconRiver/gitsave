New: React

// Component with hooks
function SaveButton() {
  const { hasChanges, filesChanged } = useGitStatus();
  const [isProcessing, setProcessing] = useState(false);
  
  return (
    <button 
      disabled={!hasChanges || isProcessing}
      onClick={handleSave}
    >
      {hasChanges 
        ? `Save Changes (${filesChanged} files)`
        : 'No changes to save'}
    </button>
  );
}
